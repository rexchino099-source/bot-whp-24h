const { 
    default: makeWASocket, 
    useMultiFileAuthState, 
    DisconnectReason,
    delay
} = require('@whiskeysockets/baileys');
const pino = require('pino');
const express = require('express');
const QRCode = require('qrcode');
const fs = require('fs');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());

const activeSockets = {};
const currentQRs = {};

// ==========================================
// 🤖 DICCIONARIO GENERAL DE COMANDOS (Compartido)
// ==========================================
async function handleCommands(sock, msg) {
    if (!msg.message || msg.key.fromMe) return;

    const messageType = Object.keys(msg.message)[0];
    const text = messageType === 'conversation' ? msg.message.conversation :
                 messageType === 'extendedTextMessage' ? msg.message.extendedTextMessage.text : '';

    const prefix = '.';
    if (!text.startsWith(prefix)) return;

    const args = text.slice(prefix.length).trim().split(/ +/);
    const command = args.shift().toLowerCase();
    const from = msg.key.remoteJid;

    switch (command) {
        case 'menu':
        case 'help':
            const menuText = 
`✨ *MENÚ PRINCIPAL DEL BOT* ✨

*📌 COMANDOS BÁSICOS:*
• *.menu* - Muestra este menú de ayuda
• *.ping* - Verifica la velocidad del bot
• *.estado* - Muestra si el servidor está activo 24/7
• *.info* - Información sobre el bot
• *.creador* - Contacto del creador del bot
• *.reglas* - Normas de uso del bot

*🎮 COMANDOS DE ENTRETENIMIENTO:*
• *.dado* - Tira un dado aleatorio (1-6)
• *.moneda* - Lanza una moneda (Cara o Cruz)
• *.azar [opcion1] [opcion2]* - Escoge al azar entre opciones

*💬 COMANDOS DE GRUPO / CHAT:*
• *.id* - Muestra tu ID o el ID del grupo
• *.hora* - Muestra la hora del servidor

_Todos los bots conectados responden a estos comandos._`;
            await sock.sendMessage(from, { text: menuText }, { quoted: msg });
            break;

        case 'ping':
            const start = Date.now();
            await sock.sendMessage(from, { text: '🏓 Calculando latencia...' }, { quoted: msg });
            const end = Date.now();
            await sock.sendMessage(from, { text: `🏓 *¡Pong!* Respuesta en *${end - start} ms*.` });
            break;

        case 'estado':
        case 'status':
            await sock.sendMessage(from, { 
                text: `🟢 *Estado:* Online 24/7 en la nube.\n⚡ *Servidor:* Operativo sin interrupciones.` 
            }, { quoted: msg });
            break;

        case 'info':
            await sock.sendMessage(from, { 
                text: `🤖 *Bot Multicuenta WhatsApp*\n\nEste bot funciona 24/7 en un servidor nube centralizado. Soporta múltiples números sincronizados.` 
            }, { quoted: msg });
            break;

        case 'creador':
        case 'owner':
            await sock.sendMessage(from, { 
                text: `👑 *Creador:* Proyecto Bot WhatsApp 24/7` 
            }, { quoted: msg });
            break;

        case 'reglas':
            await sock.sendMessage(from, { 
                text: `📜 *REGLAS DE USO:*
1. No hacer spam de comandos.
2. No usar el bot para ataques ni acoso.
3. El mal uso resultará en el bloqueo de tu número.` 
            }, { quoted: msg });
            break;

        case 'dado':
            const dado = Math.floor(Math.random() * 6) + 1;
            await sock.sendMessage(from, { text: `🎲 Lanzaste el dado y salió: *${dado}*` }, { quoted: msg });
            break;

        case 'moneda':
            const opciones = ['Cara 🪙', 'Cruz 🪙'];
            const resultado = opciones[Math.floor(Math.random() * opciones.length)];
            await sock.sendMessage(from, { text: `🪙 Lanzaste la moneda y cayó: *${resultado}*` }, { quoted: msg });
            break;

        case 'azar':
            if (args.length < 2) {
                await sock.sendMessage(from, { text: '⚠️ Escribe al menos 2 opciones. Ejemplo: `.azar pizza tacos`' }, { quoted: msg });
            } else {
                const eleccion = args[Math.floor(Math.random() * args.length)];
                await sock.sendMessage(from, { text: `🎯 De las opciones elegí: *${eleccion}*` }, { quoted: msg });
            }
            break;

        case 'hora':
            const horaActual = new Date().toLocaleTimeString('es-MX', { timeZone: 'America/Mexico_City' });
            await sock.sendMessage(from, { text: `🕒 Hora del servidor: *${horaActual}*` }, { quoted: msg });
            break;

        case 'id':
            await sock.sendMessage(from, { text: `🆔 Tu ID / Chat ID:\n\`\`\`${from}\`\`\`` }, { quoted: msg });
            break;

        default:
            break;
    }
}

// ==========================================
// 🚀 INICIALIZACIÓN DE BOTS Y SESIONES
// ==========================================
async function startBot(botId, phoneNumber = null) {
    const authFolder = `auth_info_${botId}`;
    const { state, saveCreds } = await useMultiFileAuthState(authFolder);

    const sock = makeWASocket({
        logger: pino({ level: 'silent' }),
        printQRInTerminal: false,
        auth: state,
        browser: ['Ubuntu', 'Chrome', '20.0.04']
    });

    activeSockets[botId] = sock;

    sock.ev.on('creds.update', saveCreds);

    sock.ev.on('connection.update', async (update) => {
        const { connection, lastDisconnect, qr } = update;

        if (qr) {
            currentQRs[botId] = await QRCode.toDataURL(qr);
        }

        if (connection === 'open') {
            console.log(`✅ Bot [${botId}] conectado exitosamente.`);
            delete currentQRs[botId];
        }

        if (connection === 'close') {
            const shouldReconnect = (lastDisconnect?.error?.output?.statusCode !== DisconnectReason.loggedOut);
            console.log(`⚠️ Conexión de [${botId}] cerrada. Reconectando:`, shouldReconnect);
            if (shouldReconnect) {
                startBot(botId);
            }
        }
    });

    sock.ev.on('messages.upsert', async (m) => {
        const msg = m.messages[0];
        await handleCommands(sock, msg);
    });

    if (phoneNumber && !sock.authState.creds.registered) {
        await delay(3000);
        const cleanedNumber = phoneNumber.replace(/[^0-9]/g, '');
        const code = await sock.requestPairingCode(cleanedNumber);
        return code;
    }

    return null;
}

// Cargar automáticamente sesiones guardadas al iniciar el servidor
fs.readdirSync('./').forEach(file => {
    if (file.startsWith('auth_info_')) {
        const botId = file.replace('auth_info_', '');
        console.log(`Cargando sesión guardada para: ${botId}`);
        startBot(botId);
    }
});

app.post('/pair-code', async (req, res) => {
    const { botId, phone } = req.body;
    try {
        const code = await startBot(botId, phone);
        res.json({ code });
    } catch (err) {
        res.status(500).json({ error: err.message });
    }
});

app.post('/start-qr', async (req, res) => {
    const { botId } = req.body;
    try {
        startBot(botId);
        res.json({ success: true });
    } catch (err) {
        res.status(500).json({ error: err.message });
    }
});

app.get('/get-qr/:botId', (req, res) => {
    const qr = currentQRs[req.params.botId];
    res.json({ qr: qr || null });
});

app.listen(PORT, () => {
    console.log(`🌐 Servidor Web activo en el puerto ${PORT}`);
});
